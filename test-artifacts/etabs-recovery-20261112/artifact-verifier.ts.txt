import { createHash } from 'node:crypto';
import {
  PDFDocument,
  PDFRawStream,
  PDFDict,
  PDFName,
  StandardFonts,
} from "pdf-lib";
import { stableValueHash } from "@ai-floor-planner/core";
import {
  textBoxesOverlap,
  type TextBox,
} from "@ai-floor-planner/geometry-engine";
import { XMLValidator } from "fast-xml-parser";
import * as acad from "@node-projects/acad-ts";
import type { ArtifactVerification, Floor, NativeCadVerification } from "@ai-floor-planner/core";

export function verifyCadArtifact(
  data: Uint8Array,
  format: "dwg" | "dxf",
  expected?: acad.CadDocument,
  canonicalFloors?: readonly Floor[],
  nativeEvidence?: NativeCadVerification,
): ArtifactVerification & { document?: acad.CadDocument } {
  const errors: string[] = [];
  const checks: Record<string, unknown> = {};
  const nativeValid = format === 'dwg' && nativeEvidence?.status === 'VERIFIED' && nativeEvidence.auditErrorsFound === 0 && nativeEvidence.auditErrorsFixed === 0 && nativeEvidence.insUnits === 4 && nativeEvidence.tileMode === 1 && nativeEvidence.sha256 === createHash('sha256').update(data).digest('hex');
  const nativeReaderWarnings: string[] = [];
  if (nativeEvidence && !nativeValid) errors.push('Native audit provenance or byte checksum is invalid');
  if (nativeValid) { checks.nativeAudit = nativeEvidence; checks.nativeReaderWarnings = nativeReaderWarnings; }
  try {
    if (!data.length) throw new Error("Empty CAD artifact");
    if (
      format === "dwg" &&
      Buffer.from(data.subarray(0, 6)).toString("ascii") !== "AC1032"
    )
      throw new Error("DWG header must be AC1032");
    if (format === "dxf") {
      const text = new TextDecoder("utf-8", { fatal: true }).decode(data);
      if (!/\$ACADVER\s+1\s+AC1032/.test(text) || !/0\s+EOF\s*$/.test(text))
        throw new Error("DXF version/trailer invalid");
      const sections = [
        ...text.matchAll(
          /(?:^|\r?\n)\s*0\r?\nSECTION\r?\n\s*2\r?\n([^\r\n]+)/g,
        ),
      ].map((m) => m[1]);
      if (
        !["HEADER", "TABLES", "BLOCKS", "ENTITIES", "OBJECTS"].every((s) =>
          sections.includes(s),
        )
      )
        throw new Error("Missing DXF section");
    }
    const reader =
      format === "dwg" ? new acad.DwgReader(data) : new acad.DxfReader(data);
    reader.onNotification = (_sender, event) => {
      // Native AutoCAD adds standard non-graphical view-style dictionaries unsupported
      // by acad-ts. Preserve those reader limitations explicitly after native zero-audit
      // verification; all graphic/unknown failures remain fatal.
      const viewMetadata = /^DWG VisualStyle payload is only partially mapped;/.test(event.message) || /^Unlisted object with DXF name ACDB(?:DETAIL|SECTION)VIEWSTYLE has been read as an UnknownNonGraphicalObject$/.test(event.message) || /^Entry not found Metric50\|\d+ for dictionary ACAD_(?:DETAIL|SECTION)VIEWSTYLE\|\d+$/.test(event.message);
      if (nativeValid && !event.exception && viewMetadata) nativeReaderWarnings.push(event.message);
      else if (event.exception || event.notificationType !== 0) errors.push(event.message);
    };
    const document = reader.read();
    if (!document.header || !document.entities)
      throw new Error("Missing header/entities");
    if (document.header.versionString !== "AC1032")
      errors.push("Read version is not AC1032");
    if (document.header.insUnits !== 4)
      errors.push("Read units are not millimeters (INSUNITS=4)");
    if (![5, 6].includes(document.header.splineType))
      errors.push("Invalid SPLINETYPE header");
    if (!document.header.showModelSpace)
      errors.push("Model space is not active (TILEMODE=1)");
    const entities = [...document.entities];
    if (!entities.length) errors.push("Empty entity table");
    const layers: Record<string, number> = {};
    for (const entity of entities)
      layers[entity.layer.name] = (layers[entity.layer.name] ?? 0) + 1;
    checks.layers = layers;
    checks.entityCount = entities.length;
    checks.version = document.header.versionString;
    checks.insUnits = document.header.insUnits;
    checks.tileMode = document.header.showModelSpace ? 1 : 0;
    const boxes: TextBox[] = [];
    for (const entity of entities) {
      let box: TextBox | undefined;
      if (entity instanceof acad.TextEntity) {
        const h = entity.height;
        // DXF TEXT groups 72/73 make group 11 authoritative. AutoCAD recomputes
        // group 10 after applying font metrics; inequality is valid, not corruption.
        const justified = entity.horizontalAlignment !== 0 || entity.verticalAlignment !== 0;
        const anchor = justified ? entity.alignmentPoint : entity.insertPoint;
        if (![anchor.x, anchor.y, h, entity.widthFactor].every(Number.isFinite) || h <= 0 || entity.widthFactor <= 0) { errors.push('Invalid text alignment/size'); continue; }
        const width = entity.value.length * h * .7 * entity.widthFactor;
        box = {
          x: anchor.x + (entity.horizontalAlignment === 1 ? 0 : entity.horizontalAlignment === 2 ? -width / 2 : width / 2),
          y: anchor.y + (entity.verticalAlignment === 2 ? 0 : entity.verticalAlignment === 3 ? -h / 2 : h / 2),
          width,
          height: h * 1.2,
          text: entity.value,
        };
      } else if (entity instanceof acad.DimensionAligned) {
        if (!entity.block?.name || ![...entity.block.entities].length)
          errors.push("Missing native dimension block");
        const h = entity.style.textHeight;
        const value = entity.text || String(Math.round(entity.measurement));
        box = {
          x: entity.textMiddlePoint.x,
          y: entity.textMiddlePoint.y,
          width: value.length * h * 0.7,
          height: h * 1.2,
          text: value,
        };
      }
      if (box) {
        if (boxes.some((previous) => textBoxesOverlap(box!, previous)))
          errors.push(`Annotation collision: ${box.text}`);
        boxes.push(box);
      }
    }
    checks.annotationCount = boxes.length;
    checks.annotationCollisions = errors.filter((e) =>
      e.startsWith("Annotation collision"),
    ).length;
    checks.requiredLayers = Object.fromEntries(
      [
        "A-WALL",
        "S-COLS",
        "S-BEAM",
        "A-DOOR",
        "A-WIND",
        "A-FURN",
        "A-ANNO-TEXT",
      ].map((layer) => [layer, layers[layer] ?? 0]),
    );
    if (canonicalFloors) {
      const outlines = entities.filter(
        (e): e is acad.LwPolyline =>
          e instanceof acad.LwPolyline && (e.layer.name === "A-WALL-EXT" || (canonicalFloors.some(f => f.architecture) && e.layer.name === "A-SLAB")),
      );
      if (outlines.length !== canonicalFloors.length)
        errors.push("Canonical floor outline count mismatch");
      const area = (points: readonly { x: number; y: number }[]) => {
        const origin = points[0]!;
        return (
          Math.abs(
            points.reduce((sum, p, i) => {
              const q = points[(i + 1) % points.length]!;
              return (
                sum +
                (p.x - origin.x) * (q.y - origin.y) -
                (q.x - origin.x) * (p.y - origin.y)
              );
            }, 0),
          ) / 2
        );
      };
      canonicalFloors.forEach((floor, index) => {
        const source = floor.boundary.points.filter(
          (p, i, points) =>
            !i ||
            Math.hypot(p.x - points[i - 1]!.x, p.y - points[i - 1]!.y) > 1e-9,
        );
        if (
          source.length > 1 &&
          Math.hypot(
            source[0]!.x - source.at(-1)!.x,
            source[0]!.y - source.at(-1)!.y,
          ) < 1e-9
        )
          source.pop();
        const points = outlines[index]?.vertices.map((v) => v.location) ?? [];
        if (source.length !== points.length || !outlines[index]?.isClosed) {
          errors.push(`Canonical boundary vertices mismatch: ${floor.id}`);
          return;
        }
        if (Math.abs(area(source) - area(points)) >= 0.001)
          errors.push(`Canonical floor area mismatch: ${floor.id}`);
        source.forEach((p, i) => {
          const q = source[(i + 1) % source.length]!,
            a = points[i]!,
            b = points[(i + 1) % points.length]!;
          if (
            Math.abs(q.x - p.x - (b.x - a.x)) > 1e-6 ||
            Math.abs(q.y - p.y + (b.y - a.y)) > 1e-6
          )
            errors.push(`Canonical boundary edge mismatch: ${floor.id}:${i}`);
        });
      });
      checks.canonicalFloorIds = canonicalFloors.map((f) => f.id);
      checks.canonicalFloorBoundariesVerified = !errors.length;
    }
    if (expected) {
      if (!expected.header || !expected.entities)
        throw new Error("Missing source drawing");
      const source = [...expected.entities];
      if (source.length !== entities.length)
        errors.push(
          `Entity loss: expected ${source.length}, read ${entities.length}`,
        );
      const expectedLayers: Record<string, number> = {};
      source.forEach(
        (e) =>
        (expectedLayers[e.layer.name] =
          (expectedLayers[e.layer.name] ?? 0) + 1),
      );
      for (const [layer, count] of Object.entries(expectedLayers))
        if (layers[layer] !== count)
          errors.push(
            `Layer ${layer}: expected ${count}, read ${layers[layer] ?? 0}`,
          );
      if (!nativeValid) for (const key of ["modelSpaceExtMin", "modelSpaceExtMax"] as const)
        for (const axis of ["x", "y"] as const)
          if (
            Math.abs(document.header[key][axis] - expected.header[key][axis]) >
            0.001
          )
            errors.push(`Canonical drawing bounds mismatch: ${key}.${axis}`);
      // Compare every serialized geometry control point, not merely the declared extents.
      const geometry = (e: acad.Entity) => {
        const value = e as unknown as Record<string, unknown>;
        const result: Record<string, unknown> = {
          layer: e.layer.name,
          type: e.objectName,
        };
        for (const key of [
          "startPoint",
          "endPoint",
          "center",
          "insertPoint",
          "alignmentPoint",
          "firstPoint",
          "secondPoint",
          "firstCorner", "secondCorner", "thirdCorner", "fourthCorner",
          "definitionPoint",
          "radius",
          "height",
          "value",
          "isClosed",
          "textMiddlePoint",
          "text",
        ]) {
          if (nativeValid && key === 'insertPoint' && e instanceof acad.TextEntity && (e.horizontalAlignment !== 0 || e.verticalAlignment !== 0)) continue;
          const v = value[key];
          if (v !== undefined)
            result[key] =
              v && typeof v === "object" && "x" in v
                ? {
                  x: (v as { x: number }).x,
                  y: (v as unknown as { y: number }).y,
                }
                : v;
        }
        if ("vertices" in value)
          result.vertices = Array.from(
            value.vertices as Iterable<{ location: { x: number; y: number } }>,
          ).map((v) => ({ x: v.location.x, y: v.location.y }));
        return JSON.stringify(result, (_k, v) =>
          typeof v === "number" ? Math.round(v * 1000) / 1000 : v,
        );
      };
      const originals = source.map(geometry).sort(),
        read = entities.map(geometry).sort();
      if (originals.some((e, i) => e !== read[i]))
        errors.push(
          "Serialized entity coordinates differ from canonical drawing",
        );
      for (const block of expected.blockRecords ?? []) {
        if (!block.name.startsWith("*D")) continue;
        const reopened = document.blockRecords?.get(block.name);
        const a = [...block.entities].map(geometry).sort();
        const b = reopened ? [...reopened.entities].map(geometry).sort() : [];
        if (a.length !== b.length || a.some((value, i) => value !== b[i]))
          errors.push(`Dimension block parity mismatch: ${block.name}`);
      }
      checks.canonicalBoundsParity = !errors.length;
    }
    return { isValid: !errors.length, errors, checks, document };
  } catch (error) {
    errors.push(error instanceof Error ? error.message : String(error));
    return { isValid: false, errors, checks };
  }
}

/** Reopen the uncompressed PDF object/xref graph emitted by our vector writer. */
export function verifyPdfArtifact(data: Uint8Array): ArtifactVerification {
  const errors: string[] = [];
  const text = Buffer.from(data).toString("utf8");
  const pages = [...text.matchAll(/\/Type\s*\/Page\b/g)].length;
  if (!text.startsWith("%PDF-") || !text.trimEnd().endsWith("%%EOF"))
    errors.push("Invalid PDF header/trailer");
  if (!pages) errors.push("No PDF pages");
  if (!/EXACT_SCALE|FIT_TO_PAGE_NOT_TO_SCALE|REPORT_NOT_TO_SCALE/.test(text))
    errors.push("Missing scale metadata");
  const xref = text.match(/startxref\s+(\d+)\s+%%EOF/);
  if (!xref || text.slice(Number(xref[1]), Number(xref[1]) + 4) !== "xref")
    errors.push("Invalid PDF xref offset");
  const entries = [...text.matchAll(/(\d{10}) 00000 n /g)];
  entries.forEach((m, i) => {
    if (!text.slice(Number(m[1])).startsWith(`${i + 1} 0 obj`))
      errors.push(`Invalid PDF object ${i + 1}`);
  });
  if (!entries.length) errors.push("Empty PDF object table");
  const count = Number(text.match(/\/Type \/Pages[^]*?\/Count (\d+)/)?.[1]);
  if (count !== pages) errors.push("PDF page tree count mismatch");
  for (const m of text.matchAll(/<< \/Length (\d+) >>\nstream\n/g)) {
    const start = m.index! + m[0].length;
    if (!text.slice(start + Number(m[1])).startsWith("endstream"))
      errors.push("Invalid PDF stream length");
  }
  return {
    isValid: !errors.length,
    errors,
    checks: {
      pageCount: pages,
      scaleModes: [
        ...new Set(
          [
            ...text.matchAll(
              /(?:SCALE_MODE )(EXACT_SCALE|FIT_TO_PAGE_NOT_TO_SCALE|REPORT_NOT_TO_SCALE)/g,
            ),
          ].map((m) => m[1]),
        ),
      ],
      sizeBytes: data.length,
    },
  };
}

export function verifyReportArtifact(
  data: Uint8Array,
  format: "json" | "svg",
): ArtifactVerification {
  const errors: string[] = [];
  const text = Buffer.from(data).toString("utf8");
  try {
    if (format === "json") {
      const v = JSON.parse(text);
      if (!v.revisionHash || !Array.isArray(v.floors) || !v.status)
        throw new Error("Missing canonical report provenance");
    } else {
      if (XMLValidator.validate(text) !== true)
        throw new Error("SVG XML parser rejected document");
      if (
        !/^\s*(?:<\?xml[^>]*>\s*)?<svg\b/.test(text) ||
        !text.trimEnd().endsWith("</svg>") ||
        !text.includes('xmlns="http://www.w3.org/2000/svg"') ||
        /\b(?:NaN|Infinity)\b/.test(text)
      )
        throw new Error("Invalid SVG document");
    }
  } catch (error) {
    errors.push(error instanceof Error ? error.message : String(error));
  }
  return {
    isValid: !errors.length,
    errors,
    checks: { format, sizeBytes: data.length },
  };
}

/** Reopens actual page streams and measures each glyph using the embedded standard font. */
export async function verifyComplianceArtifact(
  data: Uint8Array,
  format: "pdf" | "json",
  expectedHash: string,
): Promise<ArtifactVerification> {
  const errors: string[] = [],
    checks: Record<string, unknown> = { format };
  try {
    if (format === "json") {
      const r = JSON.parse(Buffer.from(data).toString("utf8"));
      if (
        r.schemaVersion !== 1 ||
        r.reportType !== "UDCPR_COMPLIANCE_REPORT" ||
        r.canonicalHash !== expectedHash ||
        !Number.isInteger(r.buildingVersion) ||
        r.professionalStatus !== "PROFESSIONAL_REVIEW_REQUIRED" ||
        !Array.isArray(r.rulesChecked) ||
        !r.rulesChecked.length ||
        !Array.isArray(r.sources) ||
        r.contextHash !== stableValueHash(r.regulatoryContext) ||
        !r.disclaimer?.includes("licensed Architect/Engineer")
      )
        throw new Error("Invalid compliance report provenance/context/schema");
      if (
        r.rulesChecked.some(
          (c: {
            id?: string;
            status?: string;
            elementIds?: unknown;
            sourceId?: string;
          }) =>
            !c.id ||
            !["PASS", "WARNING", "VIOLATION", "NOT_VERIFIED"].includes(
              c.status ?? "",
            ) ||
            !Array.isArray(c.elementIds) ||
            !r.sources.some((s: { id: string }) => s.id === c.sourceId),
        )
      )
        throw new Error("Invalid clause audit or source references");
      const derived = r.rulesChecked.some(
        (c: { status: string }) => c.status === "VIOLATION",
      )
        ? "VIOLATION"
        : r.rulesChecked.some(
          (c: { status: string }) =>
            c.status === "NOT_VERIFIED" || c.status === "WARNING",
        )
          ? "NOT_VERIFIED"
          : "PASS";
      if (r.status !== derived || r.overallCompliant !== (derived === "PASS"))
        throw new Error("Report status disagrees with clause findings");
      checks.canonicalHashMatch = true;
      checks.clauseCount = r.rulesChecked.length;
    } else {
      const structural = verifyPdfArtifact(data);
      if (!structural.isValid) throw new Error(structural.errors.join("; "));
      const doc = await PDFDocument.load(data, { throwOnInvalidObject: true }),
        font = await doc.embedFont(StandardFonts.Courier);
      let overlaps = 0,
        outOfBounds = 0,
        characters = 0;
      const text = Buffer.from(data).toString("ascii");
      const hashes = [...text.matchAll(/% CANONICAL_HASH ([a-f0-9]{64})/g)].map(
        (m) => m[1],
      );
      if (
        hashes.length !== doc.getPageCount() ||
        hashes.some((h) => h !== expectedHash) ||
        !text.includes(`Canonical SHA-256: ${expectedHash}`)
      )
        throw new Error("Canonical hash does not match reopened report");
      for (const page of doc.getPages()) {
        const fontDictionary = page.node
          .Resources()
          ?.lookup(PDFName.of("Font"), PDFDict)
          .lookup(PDFName.of("F1"), PDFDict);
        if (
          fontDictionary?.get(PDFName.of("BaseFont"))?.toString() !== "/Courier"
        )
          throw new Error("Unexpected report font resource");
        const contents = page.node.Contents();
        const stream = contents ? doc.context.lookup(contents) : undefined;
        if (!(stream instanceof PDFRawStream))
          throw new Error("Unsupported report content stream");
        const body = Buffer.from(stream.contents).toString("ascii");
        const commands = [
          ...body.matchAll(
            /BT \/F1 (\d+(?:\.\d+)?) Tf (-?[\d.]+) (-?[\d.]+) Td \(((?:\\.|[^\\)])*)\) Tj ET/g,
          ),
        ];
        const residue = body
          .replace(
            /BT \/F1 (\d+(?:\.\d+)?) Tf (-?[\d.]+) (-?[\d.]+) Td \(((?:\\.|[^\\)])*)\) Tj ET/g,
            "",
          )
          .replace(/^%[^\n]*(?:\n|$)/gm, "")
          .trim();
        if (!commands.length || residue)
          throw new Error("Unsupported report text operators");
        const boxes: { x: number; y: number; w: number; h: number }[] = [];
        for (const m of commands) {
          const size = Number(m[1]),
            x = Number(m[2]),
            y = Number(m[3]),
            line = m[4]!.replace(/\\([\\()])/g, "$1");
          let advance = 0;
          if (size !== 8) throw new Error("Unexpected report font size");
          for (const ch of line) {
            const w = font.widthOfTextAtSize(ch, size),
              h = font.heightAtSize(size),
              box = { x: x + advance, y: y - size * 0.3, w, h };
            advance += w;
            characters++;
            if (
              box.x < 0 ||
              box.y < 0 ||
              box.x + w > page.getWidth() ||
              box.y + h > page.getHeight()
            )
              outOfBounds++;
            if (ch !== " ") boxes.push(box);
          }
        }
        for (let i = 0; i < boxes.length; i++)
          for (let j = i + 1; j < boxes.length; j++) {
            const a = boxes[i]!,
              b = boxes[j]!;
            if (
              Math.min(a.x + a.w, b.x + b.w) - Math.max(a.x, b.x) > 0.001 &&
              Math.min(a.y + a.h, b.y + b.h) - Math.max(a.y, b.y) > 0.001
            )
              overlaps++;
          }
      }
      if (overlaps || outOfBounds)
        throw new Error(
          `Character layout invalid: ${overlaps} overlaps, ${outOfBounds} out of bounds`,
        );
      Object.assign(checks, {
        ...structural.checks,
        pdfLibReopened: true,
        canonicalHashMatch: true,
        characterBoxOverlaps: overlaps,
        outOfBounds,
        characters,
      });
    }
  } catch (e) {
    errors.push(e instanceof Error ? e.message : String(e));
  }
  return { isValid: errors.length === 0, errors, checks };
}

export async function verifyStructuralArtifact(
  data: Uint8Array,
  format: 'pdf' | 'json',
): Promise<ArtifactVerification> {
  const errors: string[] = [];
  const checks: Record<string, unknown> = { format, sizeBytes: data.length };
  try {
    if (format === 'json') {
      const v = JSON.parse(Buffer.from(data).toString('utf8'));
      if (
        !v.projectId ||
        !v.buildingId ||
        !Array.isArray(v.gridLines) ||
        !Array.isArray(v.columns) ||
        !v.summary
      ) {
        throw new Error('Invalid structural analysis model JSON');
      }
      checks.gridCount = v.gridLines.length;
      checks.columnCount = v.columns.length;
      checks.status = v.summary.overallStatus;
    } else {
      const text = Buffer.from(data.slice(0, 50)).toString('ascii');
      if (!text.startsWith('%PDF-')) errors.push('Invalid PDF header');
      const doc = await PDFDocument.load(data, { throwOnInvalidObject: true });
      const pages = doc.getPageCount();
      if (pages !== 4) errors.push(`Expected 4 structural pages, got ${pages}`);
      checks.pageCount = pages;
    }
  } catch (err) {
    errors.push(err instanceof Error ? err.message : String(err));
  }
  return {
    isValid: errors.length === 0,
    errors,
    checks,
  };
}
